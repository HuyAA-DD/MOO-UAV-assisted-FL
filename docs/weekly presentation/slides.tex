\begin{frame}[t]{Federated learning là gì?}
\relax
Nhiều thiết bị cùng học một mô hình, giữ dữ liệu huấn luyện tại chỗ.
\par\bigskip

\begin{tabularx}{\textwidth}{@{}lYY@{}}
\toprule

\begin{frame}[t]{FedAvg: ví dụ tổng hợp mô hình}
\relax
Với tập client tham gia $\mathcal S_r$, một cách lấy trung bình có trọng số là
\[
\mathbf w^{(r+1)}=\sum_{m\in\mathcal S_r}
\frac{n_m}{\sum_{j\in\mathcal S_r}n_j}\,\mathbf w_m^{(r+1)}.
\]
\begin{itemize}
\item $n_m$: số mẫu của client $m$; $\mathbf w_m^{(r+1)}$: tham số sau học cục bộ.
\item Client có nhiều mẫu hơn có trọng số lớn hơn.
\end{itemize}
\par\medskip
\textbf{Ví dụ minh họa với một tham số vô hướng:}
\[
n_1=100,\ w_1=2;\qquad n_2=300,\ w_2=4.
\]
\[
w_{\text{mới}}=\tfrac{100}{400}\cdot2+\tfrac{300}{400}\cdot4=3.5.
\]
\source{McMahan và cộng sự, AISTATS 2017, FedAvg. Ví dụ số tự xây dựng, không phải kết quả paper UAV.}
\end{frame}

\begin{frame}[t]{FedAvg: bài toán học chung}\relax
\small
Có $K$ thiết bị; thiết bị $k$ giữ tập chỉ số $\mathcal P_k$, $n_k=|\mathcal P_k|$, $n=\sum_k n_k$. Mọi thiết bị dùng cùng kiến trúc mô hình $w\in\mathbb R^d$.
\[\min_{w\in\mathbb R^d}f(w)=\frac1n\sum_{i=1}^n f_i(w)
=\sum_{k=1}^K\frac{n_k}{n}F_k(w),\qquad
F_k(w)=\frac1{n_k}\sum_{i\in\mathcal P_k}f_i(w).\]
\begin{itemize}\item $f_i(w)=\ell(x_i,y_i;w)$: lỗi dự đoán trên một mẫu.
\item $F_k$: loss trung bình cục bộ; $f$: loss trên toàn bộ dữ liệu.
\item Biến tối ưu là trọng số mô hình $w$; dữ liệu vẫn ở thiết bị.
\end{itemize}
\eqmeaning{Thiết bị nhiều mẫu có trọng số lớn hơn; mục tiêu này không mặc định cân bằng công bằng giữa các thiết bị.}
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\begin{frame}[t]{FedAvg: input và output}\relax
\small
\begin{tabularx}{\textwidth}{@{}p{2cm}Y@{}}\toprule
Thành phần & Nội dung\\\midrule
Input dữ liệu & Các tập $\mathcal P_k$ phân tán, kích thước $n_k$; server không cần nhận mẫu thô.\\
Input mô hình & Kiến trúc chung, hàm loss khả vi, khởi tạo $w_0$.\\
Siêu tham số & Tỷ lệ tham gia $C$, số epoch cục bộ $E$, minibatch $B$, learning rate $\eta$.\\
Output mỗi vòng & Mô hình toàn cục mới $w_{t+1}$ sau tổng hợp.\\
Output cuối & Mô hình $w_T$ để suy luận; chất lượng đánh giá trên dữ liệu kiểm tra riêng.\\\bottomrule\end{tabularx}
\eqmeaning{$T$ là ngân sách số vòng dùng khi triển khai mã giả; không phải số epoch cục bộ $E$.}
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\begin{frame}[t]{Objective và constraint: đọc đúng phạm vi}\relax
\small
\textbf{Objective toán học:} tối thiểu hóa empirical loss $f(w)$, thường không lồi với mạng neural.
\par\medskip
\textbf{Động cơ thiết kế:} giảm số vòng giao tiếp để đạt chất lượng mục tiêu bằng cách tăng tính toán cục bộ.
\begin{itemize}
\item Bài FedAvg không đặt một hệ ràng buộc công suất, CPU hay deadline trong bài toán $\min_w f(w)$.
\item Giả thiết thực nghiệm: tập client và dữ liệu cố định, cập nhật đồng bộ, chọn ngẫu nhiên một phần client mỗi vòng.
\item Non-IID và mất cân bằng kích thước dữ liệu là đặc tính bài toán, không phải bất đẳng thức ràng buộc.
\item Dữ liệu giữ tại client; gửi cập nhật không tự tạo bảo đảm differential privacy.
\end{itemize}
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\begin{frame}[t]{Mã giả FedAvg: phía server}\relax
\small
\begin{enumerate}
\item Khởi tạo mô hình chung $w_0$.
\item Với mỗi vòng $t=0,\ldots,T-1$:
\item Chọn ngẫu nhiên tập $S_t$ gồm $m=\max(\lfloor CK\rfloor,1)$ client.
\item Gửi $w_t$; chạy song song $w_{t+1}^k\leftarrow\operatorname{ClientUpdate}(k,w_t)$ với $k\in S_t$.
\item Tính $n_{S_t}=\sum_{k\in S_t}n_k$ và tổng hợp
\[w_{t+1}=\sum_{k\in S_t}\frac{n_k}{n_{S_t}}w_{t+1}^k.\]
\item Trả về $w_T$ khi hết ngân sách.
\end{enumerate}
\noteonly{Mã giả diễn giải Algorithm 1: quy ước làm tròn được ghi rõ. Bản v4 sửa mẫu số thành tổng số mẫu của các client được chọn.}
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\begin{frame}[t]{Mã giả FedAvg: phía client}\relax
\small
\textbf{ClientUpdate$(k,w)$}
\begin{enumerate}\item Nhận $w=w_t$ từ server; tạo các minibatch kích thước $B$ từ $\mathcal P_k$.
\item Với epoch $e=1,\ldots,E$, duyệt từng minibatch $b$:
\[w\leftarrow w-\eta\nabla\ell(w;b).\]
\item Gửi mô hình cuối $w$ về server; dữ liệu thô giữ tại chỗ.
\end{enumerate}
\eqmeaning{Mỗi client đi nhiều bước SGD từ cùng $w_t$, rồi server trung bình các điểm cuối.}
\par\medskip
$E$ lớn hoặc $B$ nhỏ làm tăng số bước local, xấp xỉ $E n_k/B$. Điều này có thể giảm giao tiếp, nhưng tăng sai lệch giữa các cập nhật khi dữ liệu khác nhau. Không nên mặc định tăng $E$ luôn tốt.
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\begin{frame}[t]{Từ FedAvg đến bài toán tài nguyên FSTPO}\relax
\small
\begin{itemize}
\item FedSGD với full batch và một bước local tổng hợp các gradient; FedAvg cho phép nhiều bước local trước khi giao tiếp.
\item Cùng điểm xuất phát giúp việc trung bình mô hình có ý nghĩa; trung bình các mạng khởi tạo độc lập có thể cho kết quả xấu.
\item Bài báo báo cáo giảm khoảng $10$--$100$ lần số vòng giao tiếp trong các thí nghiệm so với SGD đồng bộ; đây không phải bảo đảm cho mọi dữ liệu.
\item FedAvg chọn ngẫu nhiên client; FSTPO học chọn client và cấp băng thông/CPU trong mạng UAV.
\end{itemize}
\eqmeaning{Tầng học tối ưu $w$; tầng điều khiển tài nguyên chọn ai được học và với chi phí nào. Hai tầng liên quan nhưng có biến tối ưu khác nhau.}
\source{McMahan và cộng sự, 1602.05629v4, Sec. 2 và Algorithm 1 (có erratum).}
\end{frame}

\subsection{Mô hình học liên kết}

\begin{frame}[t]{Hàm mất mát cục bộ}
\relax
\begin{equation}L_m(\mathbf w)=\frac{1}{D_m}\sum_{i=1}^{D_m}\ell(\mathbf w;\mathbf x_{m,i},y_{m,i}),\tag{1}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\mathcal D_m,\ D_m$ & Tập dữ liệu của thiết bị $m$ và số mẫu trong tập.\\[.12em]
$\mathbf x_{m,i},y_{m,i}$ & Đặc trưng và nhãn của mẫu thứ $i$.\\[.12em]
$\mathbf w\in\mathbb R^d$ & Vector gồm $d$ tham số của mô hình.\\[.12em]
$\ell(\cdot)$ & Loss trên một mẫu; $L_m$ là loss trung bình tại thiết bị.
\end{tabularx}}\eqmeaning{Đo mức sai lệch của mô hình trên dữ liệu riêng của thiết bị $m$. Giá trị nhỏ hơn thể hiện mô hình khớp dữ liệu cục bộ tốt hơn.}
\paperref{II-B, (1)}
\end{frame}

\begin{frame}[t]{Hàm mất mát toàn cục}
\relax
\begin{equation}F(\mathbf w)=\sum_{m\in\mathcal M}\frac{D_m}{D_{\tot}}L_m(\mathbf w),\tag{2}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$D_{\tot}=\sum_m D_m$ & Tổng số mẫu của toàn bộ thiết bị.\\[.12em]
$D_m/D_{\tot}$ & Trọng số của thiết bị $m$ trong mục tiêu học.\\[.12em]
$F(\mathbf w)$ & Loss trung bình có trọng số trên toàn bộ dữ liệu phân tán.
\end{tabularx}}\eqmeaning{Liên kết các mục tiêu cục bộ thành một mục tiêu học chung. Thiết bị có nhiều mẫu đóng góp nhiều hơn vào loss toàn cục.}
\noteonly{Mẫu của thiết bị chưa được chọn ở một vòng vẫn thuộc tập dữ liệu định nghĩa mục tiêu toàn cục.}
\paperref{II-B, (2)}
\end{frame}

\begin{frame}[t]{Cập nhật mô hình tại thiết bị}
\relax
\begin{equation}\mathbf w_m^{(k,i)}=\mathbf w_m^{(k,i-1)}-\eta\widehat\nabla L_m(\mathbf w_m^{(k,i-1)}),\tag{3}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$i\in\{1,\ldots,I\}$ & Bước cập nhật cục bộ; $I$ là số bước SGD.\\[.12em]
$\eta$ & Tốc độ học.\\[.12em]
$\widehat\nabla L_m$ & Gradient ngẫu nhiên ước lượng từ dữ liệu cục bộ.
\end{tabularx}}\eqmeaning{Mỗi bước điều chỉnh tham số theo hướng làm giảm loss cục bộ. Chỉ thiết bị được chọn thực hiện huấn luyện.}
\[\mathbf w_m^{(k,0)}=\mathbf w^{k-1},\qquad \mathbf w_m^k=\mathbf w_m^{(k,I)}.\]
\paperref{II-B, (3)}
\end{frame}

\begin{frame}[t]{Tổng hợp mô hình toàn cục}
\relax
\begin{equation}\mathbf w^k=\frac{\sum_{u\in\mathcal U}(\sum_{m\in\mathcal M}s_{m,u}^kD_m)\mathbf w_u^k}{\sum_{u\in\mathcal U}\sum_{m\in\mathcal M}s_{m,u}^kD_m},\tag{4}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\mathbf w_u^k$ & Mô hình UAV $u$ tổng hợp từ các thiết bị trong cụm.\\[.12em]
$\sum_m s_{m,u}^kD_m$ & Tổng số mẫu của các thiết bị được UAV $u$ chọn.\\[.12em]
$\mathbf w^k$ & Mô hình chung sau tổng hợp giữa các UAV.
\end{tabularx}}\eqmeaning{Trọng số của mỗi UAV phản ánh lượng dữ liệu đang tham gia. Mô hình mới được phân phối ở vòng kế tiếp.}
\noteonly{Giả sử có ít nhất một thiết bị tham gia để mẫu số khác 0.}
\paperref{II-B, (4)}
\end{frame}

\begin{frame}[t]{Huấn luyện cục bộ: thời gian và năng lượng}
\relax

\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$I,D_m$ & Số bước học cục bộ và số mẫu dữ liệu.\\[.12em]

$f_m^k,\kappa_m$ & Tần số CPU và hệ số điện dung hiệu dụng của thiết bị.

\eqmeaning{Tăng tần số CPU làm thời gian học giảm nhưng năng lượng tính toán tăng theo bình phương tần số.}
\noteonly{Giữ nguyên công thức paper. Paper định nghĩa $D_m$ theo số mẫu nhưng ghi $c_m$ là cycles/bit; cần quy đổi mẫu sang bit để nhất quán đơn vị khi tính số.}

\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\Omega_m^k$ & Tổng kích thước mô hình và dữ liệu tác vụ.\\[.12em]

\noteonly{Tỷ số được đánh giá trên liên kết hoạt động, có tốc độ dương.}

\begin{frame}[t]{Tính toán biên trên UAV}
\relax

\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$v_m^k$ & Kích thước tác vụ thiết bị $m$ offload lên UAV.\\[.12em]

$\kappa_u$ & Hệ số điện dung hiệu dụng của CPU UAV.
\end{tabularx}}
\eqmeaning{UAV chịu thời gian và năng lượng để thực thi tác vụ được offload. Các chi phí này chỉ phát sinh khi $s_{m,u}^k=1$.}

\begin{frame}[t]{Thời gian một vòng giao tiếp}

\tag{10}
\end{equation}

\eqmeaning{Đồng bộ theo vòng tạo hiệu ứng ``thiết bị chậm'': một liên kết hoặc CPU chậm có thể kéo dài cả vòng.}

\subsection{Độ tin cậy và bài toán tối ưu}

\begin{frame}[t]{SNR tức thời}
\relax
\begin{equation}\gamma_{m,u}^k=\frac{P_m\widetilde h_{m,u}^k}{N_0b_{m,u}^k},\tag{12}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\widetilde h_{m,u}^k$ & Gain tức thời có fading, phân bố mũ với trung bình $h_{m,u}^k$.\\[.12em]
$P_m\widetilde h_{m,u}^k$ & Công suất tín hiệu nhận.\\[.12em]
$N_0b_{m,u}^k$ & Công suất nhiễu trong băng thông được cấp.
\end{tabularx}}\eqmeaning{SNR so sánh tín hiệu với nhiễu. Dù vị trí và băng thông giữ nguyên, fading vẫn có thể làm SNR biến động.}
\noteonly{Giải mã thành công nếu $\gamma_{m,u}^k\geq\gamma_{\mathrm{th}}$; ngược lại xảy ra outage.}
\paperref{II-E, (12)}
\end{frame}

\begin{frame}[t]{Xác suất truyền thành công}
\relax
\begin{equation}\Gamma_{m,u}^k=\Pr(\gamma_{m,u}^k\geq\gamma_{\mathrm{th}})=\exp\!\left(-\frac{\gamma_{\mathrm{th}}N_0b_{m,u}^k}{P_mh_{m,u}^k}\right),\tag{13}\end{equation}\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\gamma_{\mathrm{th}}$ & Ngưỡng SNR dùng để giải mã, ở thang tuyến tính trong công thức.\\[.12em]
$\Gamma_{m,u}^k$ & Xác suất thành công, nằm trong $[0,1]$.\\[.12em]
$\Gamma_{\min}$ & Ngưỡng xác suất thành công tối thiểu cần đạt.
\end{tabularx}}\eqmeaning{Chuyển biến động kênh ngẫu nhiên thành một chỉ tiêu độ tin cậy để đưa vào ràng buộc tối ưu.}
\noteonly{Với công suất phát cố định, tăng $b$ cũng tăng nhiễu trong mô hình này. $\Gamma_{\min}$ là xác suất, không phải giá trị dB.}
\paperref{II-E, (13)}
\end{frame}

\begin{frame}[t]{Utility hệ thống trong một vòng}

$\omega_1>0$ & Trọng số của lợi ích học FL.\\[.12em]
$\omega_2>0$ & Mức phạt tiêu thụ năng lượng.\\[.12em]
$\omega_3>0$ & Mức phạt thời gian hoàn thành vòng.
\end{tabularx}}\eqmeaning{Chính sách tốt cần tạo lợi ích học cao với chi phí năng lượng và độ trễ phù hợp. Tăng một thành phần có thể làm xấu thành phần khác.}
\noteonly{Các trọng số được hiệu chỉnh để các đại lượng khác đơn vị có thang số so sánh phù hợp.}
\paperref{II-F, (14)}
\end{frame}

\begin{frame}[t]{Bài toán tối ưu chung}
\relax
{\fontsize{9}{10.5}\selectfont\setlength{\jot}{0pt}
\begin{equation}
\begin{aligned}
\max_{s_{m,u}^k,b_{m,u}^k,f_m^k}\quad &\sum_{k=1}^{K}U^k\\[-.2em]
\mathrm{s.t.}\quad
\mathrm{(C1)}\;&s_{m,u}^k\in\{0,1\},&&\forall m,u,k,\\
\mathrm{(C2)}\;&\sum_{u\in\mathcal U}s_{m,u}^k\leq1,&&\forall m,k,\\
\mathrm{(C3)}\;&\sum_{m\in\mathcal M}s_{m,u}^kb_{m,u}^k\leq B_{\max},&&\forall u,k,\\
\mathrm{(C4)}\;&b_{m,u}^k\geq0,&&\forall m,u,k,\\
\mathrm{(C5)}\;&f_m^{\min}\leq f_m^k\leq f_m^{\max},&&\forall m,k,\\
\mathrm{(C6)}\;&s_{m,u}^kt_{m,u}^k\leq T_{\max},&&\forall m,u,k,\\
\mathrm{(C7)}\;&\Gamma_{m,u}^k\geq\Gamma_{\min},&&\forall m,u,k,\\
\mathrm{(C8)}\;&\Acc^K\geq\Acc_{\mathrm{target}}.
\end{aligned}\tag{15}
\end{equation}}
\eqmeaning{Tối đa tổng utility trong $K$ vòng bằng ba nhóm biến $s,b,f_m$, đồng thời đáp ứng giới hạn tài nguyên và chất lượng dịch vụ.}

\mathbf s_u^k=(\mathbf H_u^{k-\tau:k},\mathbf D_u^k,

\begin{frame}[t]{PPO có xử lý ràng buộc}
\relax
\textbf{Reward:} dùng utility $U^k$ trong (14).\\
\textbf{Cost:} đo vi phạm độ trễ và độ tin cậy.
\begin{equation}
\widehat A_{\mathrm{total}}^k=\widehat A_r^k
-\lambda_1\widehat A_{c,1}^k-\lambda_2\widehat A_{c,2}^k.
\tag{27}
\end{equation}
\begin{itemize}
\setlength{\itemsep}{.5em}
\item $\widehat A_r$: hành động tốt hơn mức kỳ vọng về reward bao nhiêu.
\item $\widehat A_{c,i}$: mức bất lợi xét theo chi phí vi phạm.
\item $\lambda_i$: tăng mức phạt khi vi phạm vượt ngưỡng cho phép.
\end{itemize}
\eqmeaning{PPO cải thiện chính sách bằng cách cân đối lợi ích với vi phạm; cập nhật được giới hạn để tránh thay đổi quá mạnh.}
\noteonly{RL xử lý chi phí tích lũy kỳ vọng; cơ chế này không trực tiếp bảo đảm accuracy (C8) ở mọi lần chạy.}
\paperref{III-C--D, (23)--(28)}
\end{frame}

\begin{frame}[t]{Chia sẻ chính sách giữa các UAV}
\relax
\[
a_u=\frac{\sum_m s_{m,u}^k}{\sum_{i\in\mathcal U}\sum_m s_{m,i}^k},
\qquad \theta_{\mathrm{glob}}=\sum_u a_u\theta_u.
\]
\[
\theta_u\leftarrow\tau\theta_{\mathrm{glob}}+(1-\tau)\theta_u.
\]
\par\medskip{\small\renewcommand{\arraystretch}{1.3}\begin{tabularx}{\textwidth}{@{}p{2.45cm}Y@{}}
$\theta_u$ & Tham số Actor của UAV $u$.\\[.12em]
$a_u$ & Trọng số theo số thiết bị đang được UAV phục vụ.\\[.12em]
$\tau$ & Hệ số cập nhật mềm; chia sẻ mỗi $E_{\mathrm{share}}$ episode.
\end{tabularx}}
\eqmeaning{Các UAV học hỏi chính sách của nhau, đồng thời giữ một phần thích nghi cục bộ.}
\noteonly{Đổi ký hiệu trọng số từ $w_u$ trong paper thành $a_u$ trên slide để tránh nhầm với mô hình FL $\mathbf w_u^k$.}
\paperref{III-D, Thuật toán 2; công thức chia sẻ chính sách}
\end{frame}

\begin{frame}[t]{Hai mô hình và hai giai đoạn hoạt động}
\relax
\renewcommand{\arraystretch}{1.45}

\toprule
 & \textbf{Mô hình FL $\mathbf w$} & \textbf{Chính sách $\theta_u$}\\

\section{Hướng mở rộng dựa trên bản thảo JNCA}
\hustpartpage{4}{HƯỚNG MỞ RỘNG}{Chọn tập thiết bị, đa mục tiêu và đa mức đánh giá}

\begin{frame}[t]{Hướng mở rộng: từ utility đến chất lượng học}\relax
\small
\textbf{Xuất phát từ FSTPO:} điều khiển $s,b,f$ để cân bằng utility FL thay thế, năng lượng và thời gian.
\par\medskip
\textbf{Bản thảo JNCA đề xuất:}
\begin{itemize}\item Giữ loss validation toàn quần thể làm mục tiêu chất lượng học.
\item Xem giá trị dữ liệu theo cả tập client: dữ liệu hiếm có thể bổ trợ, gradient khác biệt cũng có thể do nhiễu.
\item Tìm tập phương án Pareto thay vì chỉ một nghiệm ứng với bộ trọng số cố định.
\item Giảm chi phí tìm kiếm bằng đánh giá đa mức; chỉ thu thập thông tin mới khi đáng chi phí.
\end{itemize}
\noteonly{Đây là working draft: các đóng góp và lợi ích cần thực nghiệm kiểm chứng.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 1--2, 4.}
\end{frame}

\begin{frame}[t]{Mô hình mở rộng: ba mục tiêu}\relax
\small
\[\min_{\pi}\mathbf J(\pi)=\left(
\mathbb E[L_{\rm val}(w_K)],\;
\mathbb E\!\left[\sum_{k=0}^{K-1}E_k\right],\;
\mathbb E\!\left[\sum_{k=0}^{K-1}T_k\right]\right).\]
\[L_{\rm val}(w)=\sum_m p_m\frac1{|\mathcal V_m|}
\sum_{\xi\in\mathcal V_m}\ell(w;\xi),\quad \sum_m p_m=1.\]
\begin{itemize}\item Input: checkpoint, lịch sử báo cáo, kênh, giới hạn tài nguyên và pin.
\item Output: tập lịch khả thi Pareto; chọn một lịch theo ưu tiên triển khai đã định trước.
\item $p_m$ cố định theo quần thể mục tiêu; không đổi chỉ vì bỏ client khó phục vụ.
\item Validation, training và test tách biệt; validation thực tế cần quyền truy cập và tính chi phí.
\end{itemize}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 2.1, 4; (3), (27).}
\end{frame}

\begin{frame}[t]{Biến quyết định và ràng buộc mở rộng}\relax
\small
\textbf{Lõi ban đầu:} $x=(s,b^T,f^T)$; tùy chọn probing thêm $(z,b^P,f^P)$.
\[s_{m,u}\in\{0,1\},\quad\sum_u s_{m,u}\le1,\quad
0\le b^T_{m,u}\le B_u s_{m,u},\quad\sum_m b^T_{m,u}\le B_u.\]
\[f_m^{\min}s_m\le f_m^T\le f_m^{\max}s_m,\quad
1\le\sum_m s_m\le N_{\max}.\]
\[T_k\le T_{\max},\qquad B_i^{k+1}=B_i^k-E_i^k\ge B_i^{\rm reserve}.\]
\eqmeaning{Client không hoạt động nhận tài nguyên bằng 0; lịch phải thỏa khả dụng đường truyền, deadline và pin.}
\noteonly{JNCA ban đầu dùng quỹ đạo cho trước, FL đồng bộ; offloading ứng dụng độc lập của FSTPO chưa nằm trong lõi này. Kết hợp hai bài cần thống nhất phạm vi chi phí.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 3--4; (14)--(16), (24), (28).}
\end{frame}

\begin{frame}[t]{Hướng A: chọn tập client có tính bổ trợ}\relax
\small
\[\Delta V(m\mid S)=\widehat V(S\cup\{m\})-\widehat V(S).\]
\textbf{Ví dụ của bản thảo:} ba gradient vô hướng $(2,-2,1)$, trọng số bằng nhau; gradient quần thể $\bar g=1/3$.
\begin{itemize}\item Cặp $\{1,2\}$ trung bình bằng $0$, sai số bình phương $1/9$.
\item Cặp $\{1,3\}$ trung bình bằng $1.5$, sai số $49/36$.
\item Hai gradient rất khác nhau có thể bổ trợ tốt khi xét cùng nhau.
\end{itemize}
\textbf{Thuật toán:} sinh quần thể lịch khả thi $\to$ lai/đột biến tập client và tài nguyên $\to$ sửa ràng buộc $\to$ đánh giá $\to$ chọn theo Pareto và đa dạng.
\noteonly{Khớp gradient chỉ là proxy; phải kiểm tra loss validation. Giá trị biên có thể âm và không mặc định có tính submodular.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 2.2, 6; Algorithm 1.}
\end{frame}

\begin{frame}[t]{Mã giả tìm kiếm đa mức và kiểm chứng}\relax
\small
\begin{enumerate}\item Chốt snapshot, ngân sách tìm kiếm và ngân sách audit độc lập.
\item Sinh ứng viên; đánh giá tham chiếu một tập khởi tạo để hiệu chuẩn.
\item Lặp: lai/đột biến, sửa ràng buộc, tính chi phí vật lý và dự đoán loss rẻ.
\item Chọn ứng viên nâng fidelity theo lợi ích dự kiến trên chi phí; giữ tỷ lệ khám phá.
\item Cập nhật predictor và hai kho: dự đoán tạm thời, tham chiếu đã kiểm chứng.
\item Audit ứng viên cuối bằng cùng giao thức tham chiếu; tính lại quan hệ trội.
\item Chọn và thực thi một lịch theo ưu tiên đã khai báo.
\end{enumerate}
\noteonly{Chỉ kho tham chiếu dùng để báo cáo chất lượng Pareto. Rollout giả lập không cập nhật checkpoint nguồn và không được tính là vòng FL thực thi.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 7; Algorithm 2.}
\end{frame}

\begin{frame}[t]{Đề xuất kết hợp với FSTPO}\relax
\small
\textbf{Ý tưởng tổng hợp của phần trình bày, cần kiểm chứng:}
\begin{enumerate}\item Dùng actor FSTPO sinh lịch khởi tạo khả thi cho tìm kiếm tiến hóa; bổ sung lịch ngẫu nhiên để giữ đa dạng.
\item Đánh giá lịch bằng bộ ba loss--energy--time; giữ baseline FSTPO trên cùng môi trường và ngân sách.
\item Dùng predictor đa fidelity để sàng lọc và tham chiếu để xác nhận các cải thiện.
\item Sau khi lõi ổn định, thêm tuổi thông tin và probing; mọi quyết định chỉ dùng quan sát được phép.
\end{enumerate}
\noteonly{Đây không phải thuật toán đã được hai bài chứng minh. Có thể nghiên cứu scheduler tiến hóa độc lập trước; warm-start bằng FSTPO là một ablation bổ sung.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 6--10; liên hệ với Yang và cộng sự, Sec. III.}
\end{frame}

\begin{frame}[t]{Lộ trình và tiêu chí kiểm chứng}\relax
\small
\begin{enumerate}\item Bắt đầu với 2 UAV, 6 client, cùng checkpoint; đặt client dữ liệu hiếm ở kênh yếu. Liệt kê tập con với phân bổ cố định để có mốc kiểm tra.
\item So sánh chọn ngẫu nhiên, greedy kênh, FSTPO, NSGA-II và chọn tập có tính bổ trợ trên cùng thông tin/tài nguyên.
\item Chỉ thêm đa fidelity nếu profiling chứng minh đánh giá tham chiếu đủ đắt và mức rẻ có ích.
\item Báo cáo Pareto đã audit, test accuracy/loss, rare-class recall, thời gian/năng lượng, vi phạm deadline và chi phí optimizer.
\item Ablation: bỏ set-aware, bỏ multi-fidelity, bỏ probing; so sánh với cùng chi phí tìm kiếm thực tế.
\end{enumerate}
\eqmeaning{Mục tiêu là chứng minh trade-off tốt hơn sau khi tính đủ chi phí, chưa khẳng định hiệu quả trước thực nghiệm.}
\source{Information-Aware Multiobjective Evolutionary Optimization, bản thảo nghiên cứu, Sec. 9--10.}
\end{frame}
